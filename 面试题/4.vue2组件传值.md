
1.父组件应用子组件组件
```
<children :msg="message" @hhh="getValue"></children>

methods: {
  getValue(value) {
    console.log(value)
  }
}
```
2.子组件组件
```



<button @click='send'>子组件向父组件传值</button>
props:{
  msg:{
    type: String,
    default: '默认值'
  }
}
methods: {
  send() {
    this.$emit('hhh', '子组件向父组件传值')
  }
}
```
